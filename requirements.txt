# fastapi is held at 0.136.0 DELIBERATELY, not out of neglect. Read this before
# bumping it.
#
# 0.141.1 (the latest at time of writing) introduces `_IncludedRouter` in
# fastapi/routing.py, a route type that lands in `app.routes` when you call
# `app.include_router()` — which backend/main.py does twice. The pinned
# `opentelemetry-instrumentation-fastapi==0.41b0` iterates `app.routes` and reads
# `.path` off every entry, and `_IncludedRouter` has no `.path`:
#
#     AttributeError: '_IncludedRouter' object has no attribute 'path'
#     opentelemetry/instrumentation/fastapi/__init__.py:337
#
# That is not a test-only problem. Measured against a real uvicorn: every request
# returns HTTP 500. 0.136.0 is the newest release that does NOT define
# `_IncludedRouter` while still allowing `starlette>=0.46.0` unbounded — so it
# reaches a fully patched starlette without needing the whole OTel stack (5 pins)
# upgraded alongside. Bumping fastapi past 0.136.0 therefore requires upgrading
# opentelemetry-instrumentation-fastapi in the same commit, and re-running
# scripts/verify_otel.py to prove the instrumentation still works.
fastapi==0.136.0

# starlette is pinned EXPLICITLY even though fastapi pulls it in. fastapi 0.104.1
# constrained it to `<0.28.0,>=0.27.0`, so it was effectively pinned by proxy;
# 0.136.0 asks only for `>=0.46.0`, unbounded, which would leave the package that
# carries the advisories free to drift on any future install. 1.3.1 is the lowest
# version that clears all seven starlette advisories — PYSEC-2026-249 is fixed
# only in 1.3.1.
starlette==1.3.1

uvicorn[standard]==0.24.0

# aiohttp is NOT pinned here on purpose, and NOT because it is absent.
#
# It used to read `aiohttp==3.9.1`, and that one line carried 30 of the 56
# advisories `pip-audit` reports against this file. Nothing in DevGuard imports
# aiohttp — `grep -rn aiohttp --include=*.py` finds no import, and it is absent
# from `sys.modules` after `import backend.main`. But it is not removable either,
# because it arrives transitively:
#
#     chromadb -> kubernetes -> aiohttp<4.0.0,>=3.9.0
#
# So a direct pin could not remove the package; all it did was hold a dependency
# we never call at the *oldest* version that satisfied that range. Dropping the
# pin lets the resolver satisfy kubernetes's own constraint, which it does with
# 3.14.3 — above every fix version in the advisory list.
#
# If chromadb is ever cut (see SECURITY.md), kubernetes and aiohttp
# leave with it and this comment can go too.
websockets==12.0
httpx==0.25.2
