# fastapi is held at 0.136.0 DELIBERATELY, not out of neglect. Read this before
# bumping it.
#
# 0.141.1 (the latest at time of writing) introduces `_IncludedRouter` in
# fastapi/routing.py, a route type that lands in `app.routes` when you call
# `app.include_router()` — which backend/main.py does twice. The pinned
# `opentelemetry-instrumentation-fastapi==0.41b0` iterates `app.routes` and reads
# `.path` off every entry, and `_IncludedRouter` has no `.path`:
#
#     AttributeError: '_IncludedRouter' object has no attribute 'path'
#     opentelemetry/instrumentation/fastapi/__init__.py:337
#
# That is not a test-only problem. Measured against a real uvicorn: every request
# returns HTTP 500. 0.136.0 is the newest release that does NOT define
# `_IncludedRouter` while still allowing `starlette>=0.46.0` unbounded — so it
# reaches a fully patched starlette without needing the whole OTel stack (5 pins)
# upgraded alongside. Bumping fastapi past 0.136.0 therefore requires upgrading
# opentelemetry-instrumentation-fastapi in the same commit, and re-running
# scripts/verify_otel.py to prove the instrumentation still works.
fastapi==0.136.0

# starlette is pinned EXPLICITLY even though fastapi pulls it in. fastapi 0.104.1
# constrained it to `<0.28.0,>=0.27.0`, so it was effectively pinned by proxy;
# 0.136.0 asks only for `>=0.46.0`, unbounded, which would leave the package that
# carries the advisories free to drift on any future install. 1.3.1 is the lowest
# version that clears all seven starlette advisories — PYSEC-2026-249 is fixed
# only in 1.3.1.
starlette==1.3.1

uvicorn[standard]==0.24.0

# chromadb and sentence-transformers are OPTIONAL accelerators for the CWE
# retrieval store, and they are commented out here so a default install stays
# deployable on a small container. docs/INSTALLATION.md already describes them
# as optional and README's Limitations section already discloses that the
# lexical fallback is what runs.
#
# Measured on a clean venv install of this file, both are unusable anyway:
#
#   import torch                  ->  8 MB to 691 MB
#   sentence_transformers         ->  857 MB, then ImportError: cannot import
#                                     name 'cached_download' from
#                                     'huggingface_hub'
#   chromadb                      ->  AttributeError: `np.float_` was removed
#                                     in the NumPy 2.0 release
#
# sentence-transformers 2.2.2 needs huggingface_hub < 0.26 for cached_download,
# while the transformers 4.57 it pulls requires huggingface_hub >= 0.34 — a
# deadlock no pin resolves. chromadb 0.4.24 predates NumPy 2. So the 683 MB
# torch costs is spent and then thrown away: rag_store falls through to the
# same lexical embedder either way.
#
# Verified identical with and without them — same store backend ("memory"),
# same embedder ("lexical-overlap-fallback[NOT-PROD]"), same retrieval output
# for every query tried. The only difference is first-scan RSS: 913 MB with,
# 86 MB without. That is the difference between OOM and healthy on a 512 MB
# instance.
#
# To restore semantic retrieval, upgrade both to versions that work with
# NumPy 2 and current huggingface_hub, then re-enable:
# chromadb==0.4.24
# sentence-transformers==2.2.2

# setuptools is pinned because the OTel instrumentation above imports
# `pkg_resources`, which setuptools REMOVED in 81.0.0.
#
# Nothing here requests setuptools directly — chromadb pulls it in through
# `build`, unpinned, so pip takes the newest. That made this a time bomb
# rather than a stable state: installs that resolved setuptools < 81 worked,
# and the first install to resolve 81+ produces a backend that cannot import
# at all:
#
#     File "opentelemetry/instrumentation/dependencies.py", line 4
#       from pkg_resources import (
#     ModuleNotFoundError: No module named 'pkg_resources'
#
# `backend/main.py` imports FastAPIInstrumentor at module scope, so this is a
# hard boot failure, not a degraded mode — uvicorn exits before serving a
# single request. Reproduced on a clean install, which resolved setuptools
# 83.0.0.
#
# Pinning setuptools is the smaller change: the alternative is moving the
# whole opentelemetry-* set off the 1.20.0 / 0.41b0 line that
# scripts/verify_otel.py is validated against.
setuptools<81
# aiohttp is NOT pinned here on purpose, and NOT because it is absent.
#
# It used to read `aiohttp==3.9.1`, and that one line carried 30 of the 56
# advisories `pip-audit` reports against this file. Nothing in DevGuard imports
# aiohttp — `grep -rn aiohttp --include=*.py` finds no import, and it is absent
# from `sys.modules` after `import backend.main`. But it is not removable either,
# because it arrives transitively:
#
#     chromadb -> kubernetes -> aiohttp<4.0.0,>=3.9.0
#
# So a direct pin could not remove the package; all it did was hold a dependency
# we never call at the *oldest* version that satisfied that range. Dropping the
# pin lets the resolver satisfy kubernetes's own constraint, which it does with
# 3.14.3 — above every fix version in the advisory list.
#
# If chromadb is ever cut (see SECURITY.md), kubernetes and aiohttp
# leave with it and this comment can go too.
websockets==12.0
